const raw = name => fs.readFileSync(path.join(owner, name), 'utf8');
const binding = file => {
  const bytes = fs.readFileSync(file);
  return { path: path.relative(app, file).replaceAll('\\', '/'), bytes: bytes.length,
    sha256: crypto.createHash('sha256').update(bytes).digest('hex') };
};
function load(file) {
  const [bare, query] = file.split('?');
  if (query === 'raw') return fs.readFileSync(bare, 'utf8');
  if (bare.endsWith('.json')) return JSON.parse(fs.readFileSync(bare, 'utf8'));
  if (bare.endsWith('.css')) return {}; // No visual or layout claim from this check.
  if (cache.has(bare)) return cache.get(bare).exports;
  const module = { exports: {} }; cache.set(bare, module);
  const transformed = transformSync(fs.readFileSync(bare, 'utf8'), {
    sourcefile: bare, loader: 'jsx', format: 'cjs', jsx: 'automatic', target: 'node20',
  }).code;
  const localRequire = source => source.startsWith('.') ? load(path.resolve(path.dirname(bare), source)) : require(source);
  new Function('require', 'module', 'exports', transformed)(localRequire, module, module.exports);
  return module.exports;
}
const render = (Component, props = {}) => renderToStaticMarkup(React.createElement(Component, props));
const walk = (node, visit) => {
  if (!node || typeof node !== 'object') return;
  visit(node);
  for (const [key, child] of Object.entries(node)) if (!['loc', 'start', 'end'].includes(key)) {
    if (Array.isArray(child)) child.forEach(value => walk(value, visit));
    else if (child && typeof child === 'object') walk(child, visit);
  }
};

check('D33 actual quadtree literal partitions root and far parent into four nonoverlapping equal children; far cell contains exactly four supplied points', () => {
  const ast = parse(raw('GeometryFigures.jsx'), { sourceType: 'module', plugins: ['jsx'] });
  let cells;
  walk(ast, node => {
    if (node.type === 'ArrayExpression' && node.elements.length === 9 && node.elements.every(e => e?.type === 'ArrayExpression' && e.elements.length === 3 && e.elements.every(v => v.type === 'NumericLiteral')))) {
      cells = node.elements.map(e => e.elements.map(v => v.value));
    }
  });
  assert.deepEqual(cells, [[0,0,8],[0,0,4],[4,0,4],[0,4,4],[4,4,4],[4,4,2],[6,4,2],[4,6,2],[6,6,2]]);
  for (const [parent, children] of [[cells[0], cells.slice(1, 5)], [cells[4], cells.slice(5)]]) {
    const [x, y, size] = parent;
    assert.equal(children.reduce((area, c) => area + c[2] ** 2, 0), size ** 2);
    assert.deepEqual(children, [[x,y,size/2],[x+size/2,y,size/2],[x,y+size/2,size/2],[x+size/2,y+size/2,size/2]]);
  }
  const far = [[4,4],[4,5],[5,4],[5,5]];
  assert.ok(far.every(([x,y]) => x >= 4 && x < 6 && y >= 4 && y < 6));
  assert.deepEqual([0,1].map(axis => far.reduce((sum,p) => sum + p[axis],0)/4), [4.5,4.5]);
  assert.ok(render(figures.ApproximationTargets).includes('root[0,8]²'));
});
